cmake_minimum_required(VERSION 3.24)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release CACHE STRING "Build type" FORCE)
endif()
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>DLL")
project(CEEDSuperbuildConsumer LANGUAGES C CXX)

option(PALACE_WITH_LIBXSMM "Enable libXSMM" OFF)
option(PALACE_WITH_CUDA "Enable CUDA" OFF)
option(PALACE_WITH_MAGMA "Enable MAGMA" OFF)
option(PALACE_WITH_OPENMP "Enable LLVM OpenMP" OFF)
option(PALACE_CODE_PROTECTION "Enable Release protection" ON)
set(BLAS_LAPACK_LIBRARIES "" CACHE STRING "Dependencies to forward to libCEED")
if(PALACE_WITH_CUDA)
  enable_language(CUDA)
  find_package(CUDAToolkit REQUIRED)
endif()

get_filename_component(LIBCEED_SOURCE_DIR "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
include("${LIBCEED_SOURCE_DIR}/custom/ExternalLibCEED.cmake")
