cmake_minimum_required(VERSION 3.24)
project(CEEDInstalledConsumer LANGUAGES C CXX)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>DLL")
find_package(CEED CONFIG REQUIRED)
enable_testing()
set(CEED_TEST_RESOURCES "/cpu/self/ref/serial" CACHE STRING "Resources to exercise")
add_executable(consumer-c ../t011-code-protection.c)
add_executable(consumer-cxx main.cpp)
add_executable(consumer-external-jit external-jit.cpp)
target_compile_features(consumer-external-jit PRIVATE cxx_std_17)
target_link_libraries(consumer-external-jit PRIVATE CEED::ceed)
target_compile_features(consumer-c PRIVATE c_std_11)
target_compile_features(consumer-cxx PRIVATE cxx_std_17)
target_compile_definitions(consumer-cxx PRIVATE "CEED_TEST_INSTALL_ROOT=\"${CEED_TEST_INSTALL_ROOT}\"")
foreach(_target consumer-c consumer-cxx)
  target_link_libraries(${_target} PRIVATE CEED::ceed)
  if(MSVC)
    target_link_options(${_target} PRIVATE "/WHOLEARCHIVE:$<TARGET_FILE:CEED::ceed>")
  endif()
  foreach(_resource IN LISTS CEED_TEST_RESOURCES)
    string(REPLACE "/" "_" _suffix "${_resource}")
    add_test(NAME "${_target}${_suffix}" COMMAND ${_target} "${_resource}")
    set_tests_properties("${_target}${_suffix}" PROPERTIES FAIL_REGULAR_EXPRESSION ".+"
      ENVIRONMENT_MODIFICATION "CEED_DEBUG=unset:;DEBUG=unset:;DBG=unset:")
  endforeach()
endforeach()

option(CEED_TEST_EMBEDDED_JIT "Test protected CUDA JIT process reuse and dependency changes" OFF)
if(CEED_TEST_EMBEDDED_JIT)
  find_program(CEED_TEST_POWERSHELL powershell REQUIRED)
  add_test(NAME consumer-jit-cache
    COMMAND "${CEED_TEST_POWERSHELL}" -NoProfile -ExecutionPolicy Bypass -File "${CMAKE_CURRENT_SOURCE_DIR}/verify-jit-cache.ps1"
      -Probe "$<TARGET_FILE:consumer-external-jit>" -WorkDir "${CMAKE_CURRENT_BINARY_DIR}/jit-cache-tests")
endif()
